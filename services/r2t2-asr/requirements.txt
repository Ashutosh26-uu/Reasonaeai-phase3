# Reasonate AI — R2T2 ASR service (OpenAI-compatible /v1/audio/transcriptions)
#
# Every line is pinned. Install with:  pip install -r requirements.txt
#
# NOTE on PyTorch/CUDA: the R2T2 package pulls in vLLM, which in turn needs a
# torch wheel built for your CUDA runtime. That wheel is not resolvable from
# PyPI alone, so it is intentionally NOT pinned here. On a GPU host, install
# torch first from the matching index, e.g.
#   pip install torch --index-url https://download.pytorch.org/whl/cu124
# and only then install this file. See README.md ("Docker" and "GPU").

# --- HTTP service ---------------------------------------------------------
fastapi==0.115.6          # web framework; provides the OpenAI-compatible endpoints
uvicorn[standard]==0.32.1 # ASGI server used by the Dockerfile/CMD
python-multipart==0.0.20  # REQUIRED by FastAPI to parse multipart/form-data (File/Form fields)

# --- Audio / numerics -----------------------------------------------------
numpy==1.26.4             # float32 audio buffers; pinned <2 to match the vLLM/torch stack
soundfile==0.12.1         # reads the 16 kHz mono WAV that ffmpeg produces
huggingface_hub==0.27.1   # resolves and caches `netease-youdao/Confucius4-R2T2` when ASR_MODEL_PATH is a repo id

# --- Model runtime --------------------------------------------------------
# Confucius4-R2T2 (import name `r2t2`), pinned to the exact commit this service
# was written and verified against. Installing it brings the `qwen-asr[vllm]`
# dependency and, transitively, vLLM + transformers.
confucius4-r2t2 @ git+https://github.com/netease-youdao/Confucius4-R2T2.git@26d55a54ce5670cff9947a167d8ed95d569fd4d9
